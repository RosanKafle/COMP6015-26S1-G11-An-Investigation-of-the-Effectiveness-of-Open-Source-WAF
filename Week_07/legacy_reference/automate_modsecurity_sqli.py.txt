# REFERENCE ONLY; DO NOT EXECUTE. Old session redacted.
import csv
import json
import subprocess
import time
from pathlib import Path

import requests

TARGET = "http://192.168.56.20/DVWA/"
ZAP_API = "http://127.0.0.1:8080"
INPUT_FILE = Path("traffic/sqli_raw.json")
RESULT_DIR = Path("results")
RESULT_DIR.mkdir(exist_ok=True)

DETAIL_CSV = RESULT_DIR / "modsecurity_sqli_detail.csv"
SUMMARY_CSV = RESULT_DIR / "modsecurity_sqli_summary.csv"

UBUNTU_USER = "bishal"
UBUNTU_IP = "192.168.56.20"
MODSEC_LOG = "/var/log/apache2/modsec_audit.log"

PROXIES = {
    "http": "http://127.0.0.1:8080",
    "https": "http://127.0.0.1:8080",
}


def run_cmd(cmd):
    return subprocess.run(cmd, shell=True, text=True, capture_output=True)


def check_zap():
    try:
        r = requests.get(f"{ZAP_API}/JSON/core/view/version/", timeout=5)
        return r.status_code == 200, r.text
    except Exception as e:
        return False, str(e)


def check_dvwa():
    try:
        r = requests.get(TARGET, proxies=PROXIES, timeout=10, allow_redirects=False)
        return r.status_code, ""
    except Exception as e:
        return None, str(e)


def extract_method_url(msg):
    header = msg.get("requestHeader", "")
    first_line = header.splitlines()[0] if header else ""
    parts = first_line.split()
    if len(parts) >= 2:
        return parts[0], parts[1]
    return None, None


def classify(status):
    if status in [403, 406]:
        return "blocked"
    if status == 200:
        return "allowed"
    if status in [301, 302]:
        return "redirect"
    if status and 400 <= status < 500:
        return "client_error"
    if status and status >= 500:
        return "server_error"
    return "unknown"


def replay_sqli():
    with open(INPUT_FILE, "r", encoding="utf-8") as f:
        data = json.load(f)

    messages = data.get("messages", [])
    session = requests.Session()
    rows = []

    counts = {
        "total": 0,
        "blocked": 0,
        "allowed": 0,
        "redirect": 0,
        "client_error": 0,
        "server_error": 0,
        "unknown": 0,
    }

    for i, msg in enumerate(messages, start=1):
        method, url = extract_method_url(msg)
        if not url or "192.168.56.20" not in url:
            continue

        start = time.time()

        try:
            r = session.request(
                method,
                url,
                proxies=PROXIES,
                timeout=15,
                allow_redirects=False,
            )
            status = r.status_code
            error = ""
        except Exception as e:
            status = None
            error = str(e)

        elapsed = round(time.time() - start, 4)
        outcome = classify(status)

        counts["total"] += 1
        counts[outcome] += 1

        rows.append({
            "request_id": i,
            "method": method,
            "url": url,
            "status_code": status,
            "outcome": outcome,
            "response_time": elapsed,
            "error": error,
        })

        print(f"[{i}] {status} -> {outcome}")

    with open(DETAIL_CSV, "w", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(f, fieldnames=rows[0].keys())
        writer.writeheader()
        writer.writerows(rows)

    return counts


def get_remote_count(pattern):
    cmd = f"ssh {UBUNTU_USER}@{UBUNTU_IP} \"grep -c '{pattern}' {MODSEC_LOG}\""
    result = run_cmd(cmd)
    if result.returncode == 0:
        return result.stdout.strip()
    return f"ERROR: {result.stderr.strip()}"


def main():
    print("[+] Checking ZAP API...")
    zap_ok, zap_msg = check_zap()
    print(zap_msg)
    if not zap_ok:
        print("[-] ZAP API is not working. Start ZAP first.")
        return

    print("[+] Checking DVWA through ZAP...")
    status, error = check_dvwa()
    print(f"DVWA status: {status}, error: {error}")

    if not INPUT_FILE.exists():
        print(f"[-] Missing {INPUT_FILE}")
        return

    print("[+] Replaying SQLi traffic...")
    counts = replay_sqli()

    print("[+] Collecting ModSecurity log metrics from Ubuntu...")
    crs_sqli_matches = get_remote_count("942")
    access_denied = get_remote_count("Access denied")

    total = counts["total"]
    detection_rate = round((counts["blocked"] / total) * 100, 2) if total else 0
    false_negative_rate = round((counts["allowed"] / total) * 100, 2) if total else 0

    summary = {
        "mode": "modsecurity",
        "attack_type": "SQL Injection",
        "total_requests": total,
        "blocked_403_406": counts["blocked"],
        "allowed_200": counts["allowed"],
        "redirect_301_302": counts["redirect"],
        "client_errors": counts["client_error"],
        "server_errors": counts["server_error"],
        "unknown": counts["unknown"],
        "detection_rate_percent": detection_rate,
        "false_negative_rate_percent": false_negative_rate,
        "crs_942_rule_matches": crs_sqli_matches,
        "modsec_access_denied_events": access_denied,
    }

    with open(SUMMARY_CSV, "w", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(f, fieldnames=summary.keys())
        writer.writeheader()
        writer.writerow(summary)

    print("\n===== SUMMARY =====")
    for k, v in summary.items():
        print(f"{k}: {v}")

    print(f"\n[+] Detail saved: {DETAIL_CSV}")
    print(f"[+] Summary saved: {SUMMARY_CSV}")


if __name__ == "__main__":
    main()
