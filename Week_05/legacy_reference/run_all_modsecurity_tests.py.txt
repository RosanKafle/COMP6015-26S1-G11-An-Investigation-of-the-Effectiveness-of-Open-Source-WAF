def shell(cmd):
    return subprocess.run(
        cmd,
        shell=True,
        text=True,
        capture_output=True,
        timeout=10
    )


def remote_count(pattern):

    cmd = (
        f"ssh -o ConnectTimeout=5 "
        f"{UBUNTU_USER}@{UBUNTU_IP} "
        f"\"grep -c '{pattern}' {MODSEC_LOG}\""
    )

    try:
        result = shell(cmd)

        if result.returncode == 0:
            return result.stdout.strip()

        return "0"

    except subprocess.TimeoutExpired:
        print(f"[!] Timeout for pattern {pattern}")
        return "0"


def classify(status):

    if status in [403, 406]:
        return "blocked"

    if status in [200, 201, 202]:
        return "allowed"

    if status in [301, 302]:
        return "redirect"

    if status and 400 <= status < 500:
        return "client_error"

    if status and status >= 500:
        return "server_error"

    return "unknown"


def make_url(test, payload):

    base = TARGET + test["endpoint"]

    params = {
        test["param"]: payload
    }

    params.update(test["extra"])

    return base + "?" + urlencode(params)


def check_zap():

    try:
        r = requests.get(
            f"{ZAP_API}/JSON/core/view/version/",
            timeout=5
        )

        return r.status_code == 200

    except:
        return False


def run_test(attack_type, test):

    print(f"\n===== {attack_type} =====")

    rows = []

    counts = {
        "attack_type": attack_type,
        "total_requests": 0,
        "blocked": 0,
        "allowed": 0,
        "redirect": 0,
        "client_error": 0,
        "server_error": 0,
        "unknown": 0,
    }

    session = requests.Session()

    before_hits = int(remote_count(test["rule_group"]) or 0)

    for i in range(1, REQUESTS_PER_ATTACK + 1):

        payload = random.choice(test["payloads"])

        url = make_url(test, payload)

        start = time.time()

        try:

            r = session.get(
                url,
                proxies=PROXIES,
                timeout=10,
                allow_redirects=False
            )

            status = r.status_code
            error = ""

        except Exception as e:

            status = None
            error = str(e)

        elapsed = round(time.time() - start, 4)

        outcome = classify(status)

        counts["total_requests"] += 1
        counts[outcome] += 1

        rows.append({
            "attack_type": attack_type,
            "payload": payload,
            "url": url,
            "status_code": status,
            "outcome": outcome,
            "response_time": elapsed,
            "error": error,
        })

        print(
            f"[{i}/{REQUESTS_PER_ATTACK}] "
            f"{status} -> {outcome} | {payload}"
        )

        time.sleep(0.03)

    after_hits = int(remote_count(test["rule_group"]) or 0)

    counts["waf_rule_group"] = test["rule_group"]
    counts["waf_rule_hits"] = after_hits - before_hits

    total = counts["total_requests"]

    counts["detection_rate_percent"] = round(
        (counts["blocked"] / total) * 100, 2
    )

    counts["false_negative_rate_percent"] = round(
        (counts["allowed"] / total) * 100, 2
    )

    return rows, counts


def main():

    if not check_zap():

        print("[-] ZAP API NOT RUNNING")
        return

    all_rows = []
    summaries = []

    for attack_type, test in TESTS.items():

        rows, summary = run_test(
            attack_type,
            test
        )

        all_rows.extend(rows)
        summaries.append(summary)

    with open(DETAIL_CSV, "w", newline="", encoding="utf-8") as f:

        writer = csv.DictWriter(
            f,
            fieldnames=all_rows[0].keys()
        )

        writer.writeheader()
        writer.writerows(all_rows)

    with open(SUMMARY_CSV, "w", newline="", encoding="utf-8") as f:

        writer = csv.DictWriter(
            f,
            fieldnames=summaries[0].keys()
        )

        writer.writeheader()
        writer.writerows(summaries)

    with open(MATRIX_CSV, "w", newline="", encoding="utf-8") as f:

        writer = csv.writer(f)

        writer.writerow([
            "attack_type",
            "rule_group",
            "blocked",
            "allowed",
            "detection_rate_percent",
            "false_negative_rate_percent",
        ])

        for s in summaries:

            writer.writerow([
                s["attack_type"],
                s["waf_rule_group"],
                s["blocked"],
                s["allowed"],
                s["detection_rate_percent"],
                s["false_negative_rate_percent"],
            ])

    print("\n===== FINAL SUMMARY =====")

    for s in summaries:
        print(s)

    print(f"\n[+] Detail CSV: {DETAIL_CSV}")
    print(f"[+] Summary CSV: {SUMMARY_CSV}")
    print(f"[+] Matrix CSV: {MATRIX_CSV}")


if __name__ == "__main__":
    main()
